23    @Test
24    void crackByBruteForce() throws Exception {
25      // Constants
26      final String correctLogin = "admin1";
27      // State
28      final var isCracked = new Object() {
29        boolean value = false;
30      };
31      // Creating mocked data input
32      var dataIn = Mockito.mock(DataInput.class);
33      Mockito.when(dataIn.readUTF()).thenAnswer(invocation ->
34          new AuthResp(isCracked.value ? "Connection success!" : "Wrong login!")
35              .toJsonString()
36      );
37      // Creating mocked data output
38      var dataOut = Mockito.mock(FlushableDataOutput.class);
39      Mockito.doAnswer(invocation -> {
40        String userInput = invocation.getArgument(0);
41        String inputLogin = UserCredentials.fromJson(userInput).login();
42        isCracked.value = correctLogin.equals(inputLogin);
43        return null;
44      }).when(dataOut).writeUTF(Mockito.anyString());
45      // Checking correctness
46      String crackedLogin = Login.crackByBruteForce(dataIn, dataOut);
47      assertEquals(correctLogin, crackedLogin);
48    }
49  
50    @ParameterizedTest
51    @MethodSource("wasCorrect_testParams")
52    void wasCorrect(boolean expected, String message) {
53      var authResp = new AuthResp(message);

55    }
