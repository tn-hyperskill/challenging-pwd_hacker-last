16  final class ArgsTest {
17  
18    // Fields
19  
20    private static final String ipAddrStr = "127.0.0.1";
21    private static final int socketPort = 8080;
22    private static final InetAddress ipAddr = constructIpAddr();
23    private static final SocketAddress socketAddr = constructSocketAddr();
24  
25    private static final Args args = constructArgs();
26  
27    // Test methods
28  
29    @Test
30    void testParse() {
31      var executables = Stream.<Executable>of(() -> args.parse(),
32          () -> args.parse("127.0.0.1"));
33      assertAll(
34          executables.map(exec -> () -> assertThrows(Exception.class, exec)));
35    }
36  
37    @Test
38    @DisplayName("`.parse` takes exactly 2 args")
39    void parse() {
40      Args parsed = null;
41      try {
42        parsed = Args.parse(ipAddrStr, String.valueOf(socketPort));
43      } catch (UnknownHostException e) {
44        org.junit.jupiter.api.Assertions.fail(e);
45      }
46      assert parsed.socketAddr().equals(socketAddr);
47    }
48  
49    @Test
50    void socketAddr() {
51      assertEquals(args.socketAddr(), socketAddr);
52    }
53  
54    @Test
55    void ipAddr() {
56      assertEquals(args.ipAddr(), ipAddr);
57    }
58  
59    @Test
60    void port() {
61      assertEquals(args.port(), socketPort);
62    }
63  
64    // Field constructors
65  
66    private static InetAddress constructIpAddr() {
67      try {
68        return InetAddress.getByName(ipAddrStr);
69      } catch (UnknownHostException e) {
70        throw new RuntimeException(e);
71      }
72    }
73  
74    private static SocketAddress constructSocketAddr() {
75      return new InetSocketAddress(ipAddr, socketPort);
76    }
77  
78    private static Args constructArgs() {
79      try {
80        return new Args(ipAddr, socketPort);
81      } catch (UnknownHostException e) {
82        throw new RuntimeException(e);
83      }
84    }
85  }
