22    public static AutoClosableIterator<String> typicalLoginIter() {
23      final Scanner pwdSource = new Scanner(
24          Password.class.getResourceAsStream("TypicalLogins.csv"),
25          StandardCharsets.UTF_8

28      return new AutoClosableLinesIterator(pwdSource);

31    public static <O extends Flushable & DataOutput> String crackByBruteForce(
32        DataInput dataIn, O dataOut) throws Exception {
33      try (var loginIter = Login.typicalLoginIter()) {
34        while (true) {
35          // Login must be found before an exhaustion of the iterator.
36          assert loginIter.hasNext();
37          // The loop state.
38          var curLogin = loginIter.next();
39          // Brute-try the current login.
40          var authResp = Client.requestAuthentication(dataIn, dataOut,
41              new UserCredentials(curLogin, null));
42          // The loop's stop condition
43          if (Login.wasCorrect(authResp)) {
44            return curLogin;
45          }
46        }
47      }
48    }
49  
50    public static boolean wasCorrect(AuthResp authResp) {
51      return switch (authResp.message()) {
52        case "Wrong login!", "Bad request!" -> false;
53        default -> true;
54      };
55    }
